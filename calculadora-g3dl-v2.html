Configurar Google Sheets

de https://docs.google.com/spreadsheets/d/ID/edit
📋 Cómo configurar el Apps Script paso a paso
1. Abre tu Google Sheet → Extensiones → Apps Script
2. Borra el contenido y pega este código:
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.openById(data.sheetId);
  var sheet = ss.getSheetByName("Productos") || ss.getSheets()[0];
  sheet.appendRow([
    data.id, data.titulo, "", "",
    data.categoria_id, data.categoria_nombre,
    data.subcategoria_id, data.subcategoria_nombre,
    data.producto_descripcion, data.precio,
    data.estado, "", data.referencia
  ]);
  return ContentService
    .createTextOutput(JSON.stringify({ok:true}))
    .setMimeType(ContentService.MimeType.JSON);
}
function doGet(e) {
  return ContentService
    .createTextOutput(JSON.stringify({status:"ok"}))
    .setMimeType(ContentService.MimeType.JSON);
}
3. Click en Implementar → Nueva implementación
 • Tipo: Aplicación web
 • Ejecutar como: Yo
 • Acceso: Cualquier usuario
4. Copia la URL de la web app y pégala arriba.
5. Copia el ID de tu Sheet (de la URL) y pégalo también.
